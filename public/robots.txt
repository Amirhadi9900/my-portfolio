# Google's renderer obeys robots.txt when it fetches subresources, and Next serves every
# stylesheet and script under /_next/static/. A bare "Disallow: /_next/" therefore made
# Googlebot render the page unstyled and unscripted. robots.txt resolves by longest match,
# so this Allow wins for the assets while /_next/image (the optimizer) stays blocked.
Allow: /_next/static/
Disallow: /_next/
Disallow: /*.json$

# Do not add "Disallow: /image/borjian-portrait.jpg". The portrait is kept out of Image Search with
# X-Robots-Tag: noindex on that file and noimageindex on every page, and both need the
# crawler to actually fetch the URL. A Disallow line hides the instruction, not the photo.
